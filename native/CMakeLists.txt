cmake_minimum_required(VERSION 3.20)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "")
project(KasaStudioNative VERSION 0.4.1 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

set(CUBISM_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../kasapet-native/vendor" CACHE PATH "Cubism Core, Framework and Samples directory")
set(CORE_PATH "${CUBISM_SDK_ROOT}/Core")
set(FRAMEWORK_PATH "${CUBISM_SDK_ROOT}/CubismNativeFramework")
set(THIRD_PARTY "${CUBISM_SDK_ROOT}/CubismNativeSamples/Samples/OpenGL/thirdParty")
if(EXISTS "${CUBISM_SDK_ROOT}/Framework/CMakeLists.txt")
  set(FRAMEWORK_PATH "${CUBISM_SDK_ROOT}/Framework")
endif()
if(EXISTS "${CUBISM_SDK_ROOT}/Samples/OpenGL/thirdParty/glfw/CMakeLists.txt")
  set(THIRD_PARTY "${CUBISM_SDK_ROOT}/Samples/OpenGL/thirdParty")
endif()
if(NOT EXISTS "${CORE_PATH}/include/Live2DCubismCore.h")
  message(FATAL_ERROR "공식 Cubism Native SDK 경로를 -DCUBISM_SDK_ROOT=... 로 지정해 주세요.")
endif()
add_library(Live2DCubismCore STATIC IMPORTED)
set_target_properties(Live2DCubismCore PROPERTIES IMPORTED_LOCATION "${CORE_PATH}/lib/macos/${CMAKE_HOST_SYSTEM_PROCESSOR}/libLive2DCubismCore.a" INTERFACE_INCLUDE_DIRECTORIES "${CORE_PATH}/include")
set(BUILD_UTILS OFF CACHE BOOL "" FORCE)
add_subdirectory("${THIRD_PARTY}/glew/build/cmake" "${CMAKE_BINARY_DIR}/glew" EXCLUDE_FROM_ALL)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${THIRD_PARTY}/glfw" "${CMAKE_BINARY_DIR}/glfw" EXCLUDE_FROM_ALL)
set(FRAMEWORK_SOURCE OpenGL)
add_subdirectory("${FRAMEWORK_PATH}" "${CMAKE_BINARY_DIR}/Framework" EXCLUDE_FROM_ALL)
target_compile_definitions(Framework PUBLIC CSM_TARGET_MAC_GL)
target_include_directories(Framework PUBLIC "${THIRD_PARTY}/glew/include")
target_link_libraries(Framework PUBLIC Live2DCubismCore glew_s)
target_compile_options(Framework PRIVATE -w)

set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/imgui")
add_library(studio_imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp" "${IMGUI_DIR}/imgui_tables.cpp" "${IMGUI_DIR}/imgui_widgets.cpp" "${IMGUI_DIR}/backends/imgui_impl_glfw.cpp" "${IMGUI_DIR}/backends/imgui_impl_opengl2.cpp" "${IMGUI_DIR}/misc/cpp/imgui_stdlib.cpp")
target_include_directories(studio_imgui PUBLIC "${IMGUI_DIR}" "${IMGUI_DIR}/backends" "${IMGUI_DIR}/misc/cpp")
target_link_libraries(studio_imgui PUBLIC glfw "-framework OpenGL")
target_compile_options(studio_imgui PRIVATE -w)

add_library(studio_core STATIC src/state.cpp src/services.cpp src/platform_mac.mm src/tracking.cpp src/face_camera_mac.mm src/face_link_mac.mm)
set_source_files_properties(src/platform_mac.mm src/face_camera_mac.mm src/face_link_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
target_include_directories(studio_core PUBLIC src vendor)
target_compile_definitions(studio_core PRIVATE CPPHTTPLIB_THREAD_POOL_COUNT=2)
target_link_libraries(studio_core PUBLIC glfw archive "-framework Cocoa" "-framework AVFoundation" "-framework AudioToolbox" "-framework CoreAudio" "-framework ImageIO" "-framework CoreGraphics" "-framework Security")
target_compile_options(studio_core PRIVATE -Wno-deprecated-declarations)
target_link_libraries(studio_core PUBLIC "-framework Vision" "-framework CoreMedia" "-framework CoreVideo")
target_link_libraries(studio_core PUBLIC "-framework CoreImage")

add_executable(KasaStudioNative MACOSX_BUNDLE src/main.cpp src/avatar.cpp src/ui.cpp assets/AppIcon.icns)
set_source_files_properties(assets/AppIcon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
set_target_properties(KasaStudioNative PROPERTIES OUTPUT_NAME "Kasa Studio Native" MACOSX_BUNDLE_BUNDLE_NAME "Kasa Studio Native" MACOSX_BUNDLE_GUI_IDENTIFIER "com.kasa.studio.native" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/assets/Info.plist.in" MACOSX_BUNDLE_ICON_FILE AppIcon)
target_link_libraries(KasaStudioNative PRIVATE studio_core studio_imgui Framework "-framework OpenGL")
target_compile_options(KasaStudioNative PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-deprecated-declarations)
add_custom_command(TARGET KasaStudioNative POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${FRAMEWORK_PATH}/src/Rendering/OpenGL/Shaders/Standard" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/FrameworkShaders"
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/fonts"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/Paperlogy-4Regular.ttf" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/fonts/"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/assets/Anton-Regular.ttf" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/fonts/"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/assets/StudioMark.png" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/StudioMark.png"
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy "${CORE_PATH}/LICENSE.md" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/licenses/Cubism-Core.md"
  COMMAND ${CMAKE_COMMAND} -E copy "${FRAMEWORK_PATH}/LICENSE.md" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/licenses/Cubism-Framework.md"
  COMMAND ${CMAKE_COMMAND} -E copy "${THIRD_PARTY}/glfw/LICENSE.md" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/licenses/GLFW.md"
  COMMAND ${CMAKE_COMMAND} -E copy "${THIRD_PARTY}/glew/LICENSE.txt" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/licenses/GLEW.txt"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/vendor/IMGUI-LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/HTTPLIB-LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JSON-LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/assets/ANTON-OFL.txt" "${CMAKE_CURRENT_SOURCE_DIR}/assets/PAPERLOGY-OFL.txt" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/licenses/")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../data/models/hoodiecat")
  add_custom_command(TARGET KasaStudioNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../data/models/hoodiecat" "$<TARGET_BUNDLE_CONTENT_DIR:KasaStudioNative>/Resources/models/hoodiecat")
endif()
add_custom_command(TARGET KasaStudioNative POST_BUILD COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:KasaStudioNative>")

enable_testing()
add_executable(studio_tests tests/core_tests.cpp)
target_link_libraries(studio_tests PRIVATE studio_core)
target_compile_options(studio_tests PRIVATE -UNDEBUG)
add_test(NAME studio_core COMMAND studio_tests)
add_executable(studio_integration_tests tests/integration_tests.cpp)
target_link_libraries(studio_integration_tests PRIVATE studio_core)
target_compile_options(studio_integration_tests PRIVATE -UNDEBUG)
add_test(NAME studio_integrations COMMAND studio_integration_tests)
add_executable(studio_tracking_tests tests/tracking_tests.cpp)
target_link_libraries(studio_tracking_tests PRIVATE studio_core)
target_compile_options(studio_tracking_tests PRIVATE -UNDEBUG)
add_test(NAME studio_tracking COMMAND studio_tracking_tests)
add_executable(studio_phone_link_probe tests/phone_link_probe.cpp)
target_link_libraries(studio_phone_link_probe PRIVATE studio_core)
target_compile_options(studio_phone_link_probe PRIVATE -UNDEBUG)
